// Processos mais pesados, agrupados pelo .exe (o Chrome tem dezenas de processos).
static void PerfProcesses() {
    std::set<DWORD> withWindow;
    EnumWindows([](HWND w, LPARAM lp) -> BOOL { if (IsWindowVisible(w) && !GetWindow(w, GW_OWNER)) { DWORD pid = 0; GetWindowThreadProcessId(w, &pid); ((std::set<DWORD>*)lp)->insert(pid); } return TRUE; }, (LPARAM)&withWindow);
    ULONGLONG now = GetTickCount64(), dt = g_perf.prevTick ? now - g_perf.prevTick : 0;
    g_perf.prevTick = now;
    SYSTEM_INFO si; GetSystemInfo(&si);
    std::map<std::wstring, ProcUse> by;
    std::map<DWORD, std::pair<ULONGLONG, ULONGLONG>> cur;
    HANDLE snap = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (snap == INVALID_HANDLE_VALUE) return;
    PROCESSENTRY32W pe = {sizeof(pe)};
    for (BOOL ok = Process32FirstW(snap, &pe); ok; ok = Process32NextW(snap, &pe)) {
        if (pe.th32ProcessID <= 4) continue; // Ocioso e Sistema
        HANDLE h = OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, FALSE, pe.th32ProcessID);
        if (!h) continue;
        FILETIME c, e, k, u; IO_COUNTERS io = {}; PROCESS_MEMORY_COUNTERS_EX pm = {}; pm.cb = sizeof(pm);
        ULONGLONG t = GetProcessTimes(h, &c, &e, &k, &u) ? FtU(k) + FtU(u) : 0;
        ULONGLONG iob = GetProcessIoCounters(h, &io) ? io.ReadTransferCount + io.WriteTransferCount : 0;
        K32GetProcessMemoryInfo(h, (PROCESS_MEMORY_COUNTERS*)&pm, sizeof(pm));
        CloseHandle(h);
        cur[pe.th32ProcessID] = {t, iob};
        std::wstring exe = clean::LowerPath(pe.szExeFile);
        ProcUse& p = by[exe];
        p.exe = pe.szExeFile;
        p.mem += pm.PrivateUsage;
        p.window |= withWindow.count(pe.th32ProcessID) > 0;
        auto f = g_perf.prev.find(pe.th32ProcessID);
        if (dt && f != g_perf.prev.end()) {
            if (t >= f->second.first) p.cpu += (double)(t - f->second.first) / (dt * 10000.0) / si.dwNumberOfProcessors * 100.0;
            if (iob >= f->second.second) p.io += iob - f->second.second;
        }
    }
    CloseHandle(snap);
    g_perf.prev.swap(cur);
    g_perf.top.clear();
    for (auto& x : by) g_perf.top.push_back(x.second);
    int cause = g_perf.cause ? g_perf.cause : 1;
    std::sort(g_perf.top.begin(), g_perf.top.end(), [cause](const ProcUse& a, const ProcUse& b) {
        return cause == 2 ? a.mem > b.mem : cause == 3 ? a.io > b.io : a.cpu > b.cpu;
    });
    if (g_perf.top.size() > 3) g_perf.top.resize(3);
}

// Uma amostra por segundo. "Lento" sÃ³ depois de 5 s seguidos acima do limite (um pico nÃ£o conta); volta depois de 5 s calmo.
static void PerfSample() {
    FILETIME idle, kern, user;
    if (GetSystemTimes(&idle, &kern, &user)) {
        ULONGLONG i = FtU(idle), k = FtU(kern), u = FtU(user);
        if (g_perf.lastKernel) {
            ULONGLONG di = i - g_perf.lastIdle, dk = k - g_perf.lastKernel, du = u - g_perf.lastUser, tot = dk + du;
            g_perf.cpu = tot ? 100.0 * (double)(tot - di) / tot : 0;
        }
        g_perf.lastIdle = i; g_perf.lastKernel = k; g_perf.lastUser = u;
    }
    MEMORYSTATUSEX ms = {sizeof(ms)};
    if (GlobalMemoryStatusEx(&ms)) g_perf.mem = ms.dwMemoryLoad;
    if (!g_perf.q && PdhOpenQueryW(nullptr, 0, &g_perf.q) == ERROR_SUCCESS)
        PdhAddEnglishCounterW(g_perf.q, L"\\PhysicalDisk(_Total)\\% Idle Time", 0, &g_perf.diskIdle);
    if (g_perf.q && PdhCollectQueryData(g_perf.q) == ERROR_SUCCESS && g_perf.diskIdle) {
        PDH_FMT_COUNTERVALUE v;
        if (PdhGetFormattedCounterValue(g_perf.diskIdle, PDH_FMT_DOUBLE | PDH_FMT_NOCAP100, nullptr, &v) == ERROR_SUCCESS) g_perf.disk = max(0.0, min(100.0, 100.0 - v.doubleValue));
    }
    g_perf.hiCpu = g_perf.cpu >= 85 ? g_perf.hiCpu + 1 : 0;
    g_perf.hiMem = g_perf.mem >= 90 ? g_perf.hiMem + 1 : 0;
    g_perf.hiDisk = g_perf.disk >= 90 ? g_perf.hiDisk + 1 : 0;
    int cause = g_perf.hiCpu >= 5 ? 1 : g_perf.hiMem >= 5 ? 2 : g_perf.hiDisk >= 5 ? 3 : 0;
    if (cause) { g_perf.slow = true; g_perf.cause = cause; g_perf.calm = 0; }
    else if (g_perf.slow && g_perf.cpu < 70 && g_perf.mem < 85 && g_perf.disk < 70 && ++g_perf.calm >= 5) { g_perf.slow = false; g_perf.cause = 0; }
    static int every = 0;
    if (g_wd[kWPerf].mode == 1) return; // lista do Otimizar aberta: nÃ£o muda sob o mouse
    if (g_perf.slow || g_wd[kWPerf].open || ++every % 5 == 0) PerfProcesses(); // ninguÃ©m olhando: sÃ³ a cada 5 s
}

// ---------- Otimizar ----------
// SÃ³ o que comprovadamente ajuda e pode ser desfeito: fechar o que pesa (como um clique no X, nunca Ã  forÃ§a),
// liberar espaÃ§o quando o disco estÃ¡ quase cheio e reiniciar quando o PC estÃ¡ ligado hÃ¡ muitos dias.
// Nada de "limpar memÃ³ria", mexer no registro ou em serviÃ§os do Windows: isso nÃ£o acelera e pode atrapalhar.
enum { kOptClose, kOptSpace, kOptRestart };
struct OptItem { int kind; std::wstring title, sub, exe; bool on; };
static std::vector<OptItem> g_opt;

static const wchar_t* const kNeverClose[] = {L"explorer.exe", L"dwm.exe", L"csrss.exe", L"winlogon.exe", L"svchost.exe", L"lsass.exe", L"services.exe",
                                             L"sihost.exe", L"ShellExperienceHost.exe", L"StartMenuExperienceHost.exe", L"SearchHost.exe", L"Taskmgr.exe", L"Vizu.exe", L"MsMpEng.exe"};
static bool PerfClosable(const ProcUse& p) { if (!p.window) return false; for (auto n : kNeverClose) if (!_wcsicmp(p.exe.c_str(), n)) return false; return true; }
static bool DiskLow(const clean::Disk& d);

static void OptBuild() {
    g_opt.clear();
    wchar_t b[200];
    for (size_t i = 0; i < g_perf.top.size(); ++i) { // programas com janela que pesam de verdade
        const ProcUse& p = g_perf.top[i];
        if (!PerfClosable(p) || (p.cpu < 10 && p.mem < (600ull << 20))) continue;
        swprintf(b, 200, T(L"Fechar %ls", L"Close %ls", L"Cerrar %ls"), CleanProgramName(p.exe).c_str());
        wchar_t sb[160]; swprintf(sb, 160, T(L"%.0f%% da CPU Â· %ls de memÃ³ria Â· salve antes", L"%.0f%% CPU Â· %ls of memory Â· save first", L"%.0f%% de CPU Â· %ls de memoria Â· guarda antes"), p.cpu, CleanSize(p.mem).c_str());
        g_opt.push_back({kOptClose, b, sb, p.exe, g_perf.slow && g_opt.empty()}); // sÃ³ o maior culpado vem marcado, e sÃ³ se estiver lento
    }
    clean::Disk d = clean::SystemDisk();
    if (DiskLow(d)) {
        swprintf(b, 200, T(L"Liberar espaÃ§o no disco %lc:", L"Free up space on %lc:", L"Liberar espacio en %lc:"), d.letter);
        wchar_t sb[160]; swprintf(sb, 160, T(L"SÃ³ %ls livres: disco cheio deixa o Windows lento", L"Only %ls free: a full disk slows Windows down", L"Solo %ls libres: un disco lleno hace lento Windows"), CleanSize(d.free).c_str());
        g_opt.push_back({kOptSpace, b, sb, L"", true});
    }
    ULONGLONG days = GetTickCount64() / (24ull * 3600 * 1000); // "Desligar" com inicializaÃ§Ã£o rÃ¡pida nÃ£o zera isso; "Reiniciar" zera
    if (days >= 7) {
        wchar_t sb[160]; swprintf(sb, 160, T(L"Ligado hÃ¡ %llu dias Â· fecha tudo, salve antes", L"On for %llu days Â· closes everything, save first", L"Encendido hace %llu dÃ­as Â· cierra todo, guarda antes"), days);
        g_opt.push_back({kOptRestart, T(L"Reiniciar o PC", L"Restart the PC", L"Reiniciar el PC"), sb, L"", false});
    }
}

static void OptApply(HWND owner) {
    int closing = 0; bool space = false, restart = false;
    for (auto& o : g_opt) {
        if (!o.on) continue;
        if (o.kind == kOptClose) { std::wstring exe = o.exe; std::thread([exe]() { clean::CloseProgram(exe); }).detach(); ++closing; } // como um clique no X
        else if (o.kind == kOptSpace) space = true;
        else restart = true;
    }
    if (closing) {
        wchar_t b[120]; swprintf(b, 120, T(L"Pedindo para fechar %d programa(s). Quem tiver algo sem salvar vai perguntar.", L"Asking %d app(s) to close. Apps with unsaved work will ask first.", L"Pidiendo cerrar %d programa(s). Los que tengan algo sin guardar preguntarÃ¡n."), closing);
        ShowToast(T(L"Otimizar", L"Optimize", L"Optimizar"), b, 4000, 1);
    }
    if (space) OpenCleanWindow();
    if (restart && MessageBoxW(owner, T(L"Reiniciar o PC agora? Salve o que estiver aberto antes.", L"Restart the PC now? Save your open work first.", L"Â¿Reiniciar el PC ahora? Guarda lo que tengas abierto antes."),
                               L"Vizu", MB_YESNO | MB_ICONQUESTION | MB_DEFBUTTON2 | MB_TOPMOST) == IDYES) {
        HANDLE tok; // reinÃ­cio normal: os programas podem pedir para salvar (sem EWX_FORCE)
        if (OpenProcessToken(GetCurrentProcess(), TOKEN_ADJUST_PRIVILEGES | TOKEN_QUERY, &tok)) {
            TOKEN_PRIVILEGES tp = {1}; LookupPrivilegeValueW(nullptr, SE_SHUTDOWN_NAME, &tp.Privileges[0].Luid); tp.Privileges[0].Attributes = SE_PRIVILEGE_ENABLED;
            AdjustTokenPrivileges(tok, FALSE, &tp, 0, nullptr, nullptr); CloseHandle(tok);
        }
        ExitWindowsEx(EWX_REBOOT, SHTDN_REASON_MAJOR_OTHER | SHTDN_REASON_FLAG_PLANNED);
    }
}

static std::wstring PerfCulprit() {
    if (!g_perf.slow || g_perf.top.empty()) return L"";
    const ProcUse& p = g_perf.top[0];
    wchar_t b[160];
    std::wstring n = CleanProgramName(p.exe);
    if (g_perf.cause == 2) swprintf(b, 160, T(L"%ls Â· %ls de memÃ³ria", L"%ls Â· %ls of memory", L"%ls Â· %ls de memoria"), n.c_str(), CleanSize(p.mem).c_str());
    else if (g_perf.cause == 3) swprintf(b, 160, T(L"%ls Â· usando o disco", L"%ls Â· using the disk", L"%ls Â· usando el disco"), n.c_str());
    else swprintf(b, 160, T(L"%ls Â· %.0f%% da CPU", L"%ls Â· %.0f%% of CPU", L"%ls Â· %.0f%% de la CPU"), n.c_str(), p.cpu);
    return b;
}

static bool DiskLow(const clean::Disk& d) { return d.total && ReadDword(L"DiskWarn", 1) && d.free * 100 < d.total * min(50ul, max(1ul, (unsigned long)ReadDword(L"DiskLimit", 10))); }

static void WidgetPresent(Widget& wd, HBITMAP bmp, LONG px, LONG py, UINT Wt, UINT Ht) {
    wd.w = Wt; wd.h = Ht;
    HDC screen = GetDC(nullptr), mem = CreateCompatibleDC(screen);
    HGDIOBJ old = SelectObject(mem, bmp);
    POINT pos = {px, py}, zero = {0, 0};
    SIZE size = {(LONG)Wt, (LONG)Ht};
    BLENDFUNCTION bf = {AC_SRC_OVER, 0, 255, AC_SRC_ALPHA};
    UpdateLayeredWindow(wd.wnd, screen, &pos, &size, mem, &zero, 0, &bf, ULW_ALPHA);
    SelectObject(mem, old); DeleteDC(mem); ReleaseDC(nullptr, screen); DeleteObject(bmp);
}

// Bateria anexada Ã  barra de tarefas: Ã­cone e porcentagem de cada aparelho numa linha, como o relÃ³gio anexado.
static void BatRenderDocked(Widget& wd, const RECT& tb, const std::vector<BatDev>& bat) {
    float s = (tb.bottom - tb.top) / 48.0f;
    bool dk = TaskbarDark();
    D2D1_COLOR_F text = dk ? ui::Rgba(0xFFFFFF, 0.95f) : ui::Rgba(0x1D1D1F), text2 = dk ? ui::Rgba(0xFFFFFF, 0.62f) : ui::Rgba(0x000000, 0.58f);
    D2D1_COLOR_F hi = g_cfg.themeMode == ui::kThemeWindows ? g_theme.accent : ui::Rgba(0xFF6633);
    size_t n = min(bat.size(), (size_t)4);
    UINT Wt = (UINT)((n ? 14 + n * 70 : 120) * s), Ht = (UINT)((tb.bottom - tb.top) * 0.86f);
    fmt::Canvas cv(Wt, Ht);
    if (!cv.rt) return;
    ID2D1RenderTarget* rt = cv.rt;
    rt->Clear(D2D1::ColorF(0, 0));
    rt->SetTextAntialiasMode(D2D1_TEXT_ANTIALIAS_MODE_GRAYSCALE);
    ui::Painter p(rt, g_dw, g_theme, s);
    p.Fill(D2D1::RectF(0, 0, (float)Wt, (float)Ht), 8 * s, wd.hot || wd.open ? ui::Rgba(dk ? 0xFFFFFF : 0x000000, 0.08f) : ui::Rgba(0x000000, 0.006f));
    float cy = Ht / 2.0f, x = 10 * s;
    if (!n) ClText(p, T(L"Sem aparelhos", L"No devices", L"Sin dispositivos"), D2D1::RectF(x, cy - 9 * s, (float)Wt - 6 * s, cy + 9 * s), 12, text2);
    for (size_t i = 0; i < n; ++i) {
        bool low = bat[i].pct <= 15;
        p.Feather(BatIcon(bat[i].name), D2D1::RectF(x, cy - 9 * s, x + 18 * s, cy + 9 * s), 17 * s, low ? hi : text2);
        wchar_t b[16]; swprintf(b, 16, L"%d%%", bat[i].pct);
        ClText(p, b, D2D1::RectF(x + 22 * s, cy - 9 * s, x + 66 * s, cy + 9 * s), 12.5f, low ? hi : text, DWRITE_FONT_WEIGHT_SEMI_BOLD);
        x += 70 * s;
    }
    HBITMAP bmp = nullptr;
    if (FAILED(cv.Finish(&bmp)) || !bmp) return;
    LONG px = max(tb.left + 4, min(tb.right - (LONG)Wt - 4, wd.ax)), py = tb.top + ((tb.bottom - tb.top) - (LONG)Ht) / 2;
    WidgetPresent(wd, bmp, px, py, Wt, Ht);
}

    { std::lock_guard<std::mutex> lk(g_batMx); bat = g_bat; }

// BotÃ£o direito: anexar/desanexar (desempenho) e ocultar.
static void WidgetDock(int k, bool on);
static void WidgetMenu(int k, POINT pt) {
    HMENU m = CreatePopupMenu();
    enum { kDock = 1, kHide };
    if (WCanDock(k)) AppendMenuW(m, MF_STRING, kDock, g_wd[k].docked ? T(L"Desanexar da barra de tarefas", L"Detach from the taskbar", L"Desanclar de la barra de tareas")
                                                                    : T(L"Anexar Ã  barra de tarefas", L"Attach to the taskbar", L"Anclar a la barra de tareas"));
    AppendMenuW(m, MF_STRING, kHide, T(L"Ocultar widget", L"Hide widget", L"Ocultar widget"));
    SetForegroundWindow(g_wd[k].wnd);
    int cmd = TrackPopupMenu(m, TPM_RETURNCMD | TPM_RIGHTBUTTON, pt.x, pt.y, 0, g_wd[k].wnd, nullptr);
    DestroyMenu(m);
    if (cmd == kDock) WidgetDock(k, !g_wd[k].docked);
    else if (cmd == kHide) { WriteDword(WKey(k), 0); PostMessageW(g_wd[k].wnd, WM_APP + 70, k, 0); } // fecha fora do laÃ§o do menu
}

static int WidgetKind(HWND h) { for (int k = 0; k < kWCount; ++k) if (g_wd[k].wnd == h) return k; return -1; }

static void WidgetEnable(int k, bool on);
static LRESULT CALLBACK WidgetProc(HWND h, UINT msg, WPARAM wp, LPARAM lp) {
    int k = WidgetKind(h);
    if (k < 0) return DefWindowProcW(h, msg, wp, lp);
    Widget& wd = g_wd[k];
    switch (msg) {
    case WM_TIMER:
        if (wp == 1) { // 1 s
            if (k == kWPerf) PerfSample();
            static int batTick = 0;
            if (k == kWBat && ++batTick % 60 == 0) BatRefresh(h);
            if (wd.docked && !wd.open) SetWindowPos(h, HWND_TOPMOST, 0, 0, 0, 0, SWP_NOMOVE | SWP_NOSIZE | SWP_NOACTIVATE);
            if (wd.open && wd.leaveAt && GetTickCount64() > wd.leaveAt) { wd.open = false; wd.leaveAt = 0; wd.hot = 0; wd.mode = 0; }
            static int tick = 0;
            if (k == kWPerf || wd.open || ++tick % 30 == 0) WidgetRender(k); // disco: muda devagar
            // Re-arm timer with visibility awareness
            bool visible = IsWindowVisible(h) && !IsIconic(h);
            // Check if fullscreen app is running (pause condition)
            HWND fg = GetForegroundWindow();
            bool fullscreen = false;
            if (fg && fg != h) {
                RECT fgRect;
                MONITORINFO mi = {sizeof(mi)};
                if (GetWindowRect(fg, &fgRect) && GetMonitorInfoW(MonitorFromWindow(fg, MONITOR_DEFAULTTONEAREST), &mi)) {
                    if (fgRect.left <= mi.rcMonitor.left && fgRect.top <= mi.rcMonitor.top &&
                        fgRect.right >= mi.rcMonitor.right && fgRect.bottom >= mi.rcMonitor.bottom) {
                        fullscreen = true;
                    }
                }
            }
            WidgetSetTimer(h, k, visible && !fullscreen);
        }
        return 0;
    case WM_MOUSEMOVE: {
        POINT pt = {GET_X_LPARAM(lp), GET_Y_LPARAM(lp)};
        if (wd.pressed && !wd.dragging && (abs(pt.x - wd.down.x) > 4 || abs(pt.y - wd.down.y) > 4)) {
            wd.dragging = true; wd.open = false; WidgetRender(k);
            ReleaseCapture(); SendMessageW(h, WM_NCLBUTTONDOWN, HTCAPTION, 0); // arrastar
            wd.pressed = wd.dragging = false;
            return 0;
        }
        if (k == kWBat && wd.docked) { // anexada: sÃ³ realÃ§a; nÃ£o abre o cartÃ£o em cima da barra de tarefas
            if (!wd.hot) { wd.hot = 1; WidgetRender(k); }
            TRACKMOUSEEVENT te = {sizeof(te), TME_LEAVE, h, 0}; TrackMouseEvent(&te);
            return 0;
        }
        int hot = ui::Hit(wd.rBtn, pt) ? 1 : ui::Hit(wd.rClose, pt) ? 2 : ui::Hit(wd.rOpt, pt) ? 3 : ui::Hit(wd.rApply, pt) ? 4 : ui::Hit(wd.rBack, pt) ? 5 : 0;
        for (size_t i = 0; i < wd.rItems.size(); ++i) if (ui::Hit(wd.rItems[i], pt)) hot = 10 + (int)i;
        bool changed = hot != wd.hot || !wd.open;
        wd.hot = hot; wd.leaveAt = 0;
        if (!wd.open) { wd.open = true; if (k == kWPerf) PerfProcesses(); SetWindowPos(h, HWND_TOPMOST, 0, 0, 0, 0, SWP_NOMOVE | SWP_NOSIZE | SWP_NOACTIVATE); } // aberto fica por cima do outro widget
        if (changed) WidgetRender(k);
        TRACKMOUSEEVENT te = {sizeof(te), TME_LEAVE, h, 0}; TrackMouseEvent(&te);
        return 0;
    }
    case WM_MOUSELEAVE: if (k == kWBat && wd.docked) { wd.hot = 0; WidgetRender(k); return 0; }
        wd.leaveAt = GetTickCount64() + (wd.mode ? 2500 : 400); wd.hot = 0; return 0; // fecha com um respiro (mais longo no Otimizar)
    case WM_SETCURSOR: SetCursor(LoadCursor(nullptr, wd.hot ? IDC_HAND : IDC_ARROW)); return TRUE;
    case WM_LBUTTONDOWN: wd.pressed = true; wd.down = {GET_X_LPARAM(lp), GET_Y_LPARAM(lp)}; SetCapture(h); return 0;
    case WM_LBUTTONUP: {
        bool was = wd.pressed; wd.pressed = false; ReleaseCapture();
        if (!was) return 0;
        POINT pt = {GET_X_LPARAM(lp), GET_Y_LPARAM(lp)};
        for (size_t i = 0; i < wd.rItems.size() && i < g_opt.size(); ++i) if (ui::Hit(wd.rItems[i], pt)) { g_opt[i].on = !g_opt[i].on; WidgetRender(k); return 0; }
        if (ui::Hit(wd.rOpt, pt)) { OptBuild(); wd.mode = 1; WidgetRender(k); return 0; }
        if (ui::Hit(wd.rBack, pt)) { wd.mode = 0; WidgetRender(k); return 0; }
        if (ui::Hit(wd.rApply, pt)) { wd.mode = 0; wd.open = false; WidgetRender(k); OptApply(h); return 0; }
        if (k == kWBat) {
            if (wd.docked || ui::Hit(wd.rBtn, pt)) BatRefresh(h); // anexada: clique atualiza
            return 0;
        }
        if (ui::Hit(wd.rClose, pt) && !g_perf.top.empty()) {
            std::wstring exe = g_perf.top[0].exe;
            std::thread([exe]() { clean::CloseProgram(exe); }).detach(); // pede para fechar; nunca forÃ§a
        } else if (ui::Hit(wd.rBtn, pt) || (k == kWDisk && !wd.open)) OpenCleanWindow();
        return 0;
    }
    case WM_MOVING: {
        RECT tb;
        if (!(wd.docked && TaskbarRect(&tb))) break;
        RECT* r = (RECT*)lp; LONG ww = r->right - r->left, hh = r->bottom - r->top;
        r->top = tb.top + ((tb.bottom - tb.top) - hh) / 2; r->bottom = r->top + hh;
        if (r->left < tb.left + 4) { r->left = tb.left + 4; r->right = r->left + ww; }
        if (r->right > tb.right - 4) { r->right = tb.right - 4; r->left = r->right - ww; }
        return TRUE;
    }
    case WM_RBUTTONUP: { POINT pt; GetCursorPos(&pt); wd.open = false; wd.mode = 0; WidgetRender(k); WidgetMenu(k, pt); return 0; }
    case WM_APP + 70: WidgetEnable((int)wp, false); return 0;
    case WM_APP + 71: WidgetRender(k); return 0; // leitura nova (uso de IA ou bateria)
    case WM_EXITSIZEMOVE: {
        RECT r; GetWindowRect(h, &r);
        if (wd.docked) { wd.ax = r.left; WriteDword(WDockXKey(k), (DWORD)r.left); return 0; }
        float M = 14 * GetDpiForWindow(h) / 96.0f;
        wd.ax = r.left + (LONG)M; wd.ay = r.top + (LONG)M;
        WriteDword(WKeyXY(k, false).c_str(), (DWORD)wd.ax); WriteDword(WKeyXY(k, true).c_str(), (DWORD)wd.ay);
        return 0;
    }
    case WM_DISPLAYCHANGE: case WM_SETTINGCHANGE: case WM_DPICHANGED: WidgetRender(k); break;
    }
    return DefWindowProcW(h, msg, wp, lp);
}

static bool WidgetOn(int k) { return !g_lite && ReadDword(WKey(k), 0) != 0; }

static void WidgetSetTimer(HWND h, int k, bool visible) {
    if (!h) return;
    if (visible) {
        // Perf: 1s for real-time; Bat/Disk: 60s is enough
        UINT interval = (k == kWPerf) ? 1000 : 60000;
        SetTimer(h, 1, interval, nullptr);
    } else {
        KillTimer(h, 1);
    }
}

static void WidgetEnable(int k, bool on) {
    if (k == kWAi) { NotchEnable(on && !g_lite); return; }
    Widget& wd = g_wd[k];
    if (!on || g_lite) { if (wd.wnd) { KillTimer(wd.wnd, 1); DestroyWindow(wd.wnd); wd.wnd = nullptr; } return; }
    if (wd.wnd) { WidgetRender(k); return; }
    static bool reg = false;
    if (!reg) { WNDCLASSW wc = {}; wc.lpfnWndProc = WidgetProc; wc.hInstance = g_inst; wc.lpszClassName = L"VizuWidget"; RegisterClassW(&wc); reg = true; }
    wd = Widget();
    wd.wnd = CreateWindowExW(WS_EX_TOOLWINDOW | WS_EX_TOPMOST | WS_EX_LAYERED | WS_EX_NOACTIVATE, L"VizuWidget", L"Vizu", WS_POPUP, 0, 0, 10, 10, nullptr, nullptr, g_inst, nullptr);
    DWORD x = ReadDword(WKeyXY(k, false).c_str(), 0x80000000), y = ReadDword(WKeyXY(k, true).c_str(), 0x80000000);
    POINT pt = {(LONG)x, (LONG)y};
    if (x == 0x80000000 || !MonitorFromPoint(pt, MONITOR_DEFAULTTONULL)) { // canto de cima, Ã  direita; o de desempenho logo abaixo
        MONITORINFO mi = {sizeof(mi)}; GetMonitorInfoW(MonitorFromPoint({0, 0}, MONITOR_DEFAULTTOPRIMARY), &mi);
        float s = GetDpiForSystem() / 96.0f;
        size_t n = ReadDword(L"DiskAll", 0) ? max((size_t)1, clean::AllDisks().size()) : 1; // o de desempenho fica logo abaixo do de disco
        float below = WidgetOn(kWDisk) ? 20 + 56 + (n - 1) * 30 + 16 : 20;
        if (k >= kWAi) below = (WidgetOn(kWPerf) ? below + 76 : below); // a bateria fica embaixo dos outros
        pt.x = mi.rcWork.right - (LONG)(240 * s); pt.y = mi.rcWork.top + (LONG)((k == kWDisk ? 20 : below) * s);
    }
    wd.ax = pt.x; wd.ay = pt.y;
    RECT tb;
    if (WCanDock(k) && ReadDword(WDockKey(k), 0) && TaskbarRect(&tb)) {
        wd.docked = true;
        DWORD dx = ReadDword(WDockXKey(k), 0x80000000);
        wd.ax = dx == 0x80000000 ? tb.right - (LONG)((k == kWBat ? 560 : 360) * GetDpiForSystem() / 96) : (LONG)dx; // perto da bandeja, sem cobrir o relÃ³gio do Windows
    }
    if (k == kWPerf) PerfSample();
    if (k == kWBat) BatRefresh(wd.wnd);
    WidgetRender(k);
    ShowWindow(wd.wnd, SW_SHOWNOACTIVATE);
    WidgetSetTimer(wd.wnd, k, true); // visible = true initially
}

static void WidgetDock(int k, bool on) {
    Widget& wd = g_wd[k];
    RECT tb;
    if (on && !TaskbarRect(&tb)) { MessageBoxW(wd.wnd, T(L"A barra de tarefas precisa estar na horizontal (embaixo ou em cima).", L"The taskbar must be horizontal (bottom or top).", L"La barra de tareas debe estar en horizontal."), L"Vizu", MB_OK | MB_ICONINFORMATION); return; }
    WriteDword(WDockKey(k), on);
    WidgetEnable(k, false); WidgetEnable(k, true); // recria na posiÃ§Ã£o certa
}

static void WidgetsStart() { for (int k = 0; k < kWCount; ++k) WidgetEnable(k, WidgetOn(k)); }
